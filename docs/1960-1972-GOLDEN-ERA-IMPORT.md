```bash
# Verify era distribution
sqlite3 db/local.db "
  SELECT
    CASE
      WHEN year BETWEEN 1960 AND 1964 THEN '1960-1964'
      WHEN year BETWEEN 1965 AND 1967 THEN '1965-1967'
      WHEN year BETWEEN 1968 AND 1970 THEN '1968-1970'
      WHEN year BETWEEN 1971 AND 1972 THEN '1971-1972'
      ELSE 'Other'
    END as era,
    COUNT(*) as count
  FROM cars_for_sale
  WHERE year BETWEEN 1960 AND 1972
  GROUP BY era
  ORDER BY era;
"
```
